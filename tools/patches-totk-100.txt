# Code patches for TotK 1.0.0 main (tools/wordpatches.py, gen.py --patches).
#
# Sites and replacement words are those of totk-pc-recomp's recomp/patches-v121.txt (MIT), which took them from
# MacNativeGames games/totk/runtime/game_patches.c (MIT, Copyright (c) 2026 MacNativeGames contributors); the 1.0.0
# offsets are MacNativeGames' game_version.h mapping of each 1.2.1 site. Each original word is checked against this
# module by the generator (all of them matched).
#
# The patch page (wordpatches.PAGE_ADDR) holds what the sets read: +0 the 30 Hz ratio (seconds * 30) and +4 the frame step in
# seconds (realtime; runtime/totk_frame.c writes both at every present), +16 the scene size {w, h} as two floats (scene).

set scene    SAVAGEWIND_PATCH_SCENE    0
set fps60    SAVAGEWIND_PATCH_FPS60    1
set realtime SAVAGEWIND_PATCH_REALTIME 1
set bounce   SAVAGEWIND_PATCH_BOUNCE   1
set cloth    SAVAGEWIND_PATCH_CLOTH    1

# Scene render size: the renderer's +0xcc4 / +0xcc8, which size every scene render target, take two floats from page +16
# instead of the graphics parameters' docked 1600x900 (1.2.1 main+0xcc3f68..74).
site scene main c630bc 5e21d900 @page:x9    # scvtf s0, s8 -> adrp x9, page
site scene main c630c4 bd0cc660 2d420520    # str s0, [x19, #0xcc4] -> ldp s0, s1, [x9, #16]
site scene main c630c8 5e21d921 bd0cc660    # scvtf s1, s9 -> str s0, [x19, #0xcc4]

# 60 fps: the adaptive present-interval controller (1.0.0 main+0x8f6bec) gets the minimum interval 1 (shipped: 2 = 30 fps).
site fps60 main 8f6c60 6b08015f 7100055f   # clamp: cmp w10, #1 (was cmp w10, w8 = minimum)
site fps60 main 8f6c64 1a89b10a 1a9fa52a   # clamp: csinc w10, w9, wzr, ge (desired = max(desired, 1))
site fps60 main 8f6c88 b9401a8a d503201f   # lower trigger: nop the stored-candidate load
site fps60 main 8f6d18 6b08015f 7100055f   # raise path: cmp w10, #1
site fps60 main 8f6d28 1a88c1ac 1a9fc5ac   # raise path: csinc w12, w13, wzr, gt (next lower candidate floors at 1)
site fps60 main 8f6d7c b9400288 52800028   # lower path: mov w8, #1 (minimum interval 1)
site fps60 main 8f6d98 2a0a03f5 1a9fa555   # lower path: csinc w21, w10, wzr, ge (result never below 1)

# Real-time step: the VFR step (1.0.0 main+0x7a83e4) and the global ratio after the controller call (main+0x8f68b8) read
# ratio and seconds from the patch page (+0 ratio = seconds * 30, +4 seconds) instead of interval * 0.5 and interval / 60.
site realtime main 7a83f4 1e220020 @page:x9   # scvtf s0, w1 -> adrp x9, page
site realtime main 7a83f8 52a84e08 bd400120   # mov w8, #60.0 -> ldr s0, [x9] (ratio)
site realtime main 7a83fc 1e270101 bd400521   # fmov s1, w8 -> ldr s1, [x9, #4] (seconds)
site realtime main 7a8400 1e2c1002 d503201f   # fmov s2, #0.5 -> nop
site realtime main 7a840c 1e211801 d503201f   # fdiv s1, s0, s1 -> nop
site realtime main 7a8410 1e220800 d503201f   # fmul s0, s0, s2 -> nop
site realtime main 8f68b8 bd425e60 @page:x9   # ldr s0, [x19, #0x25c] -> adrp x9, page
site realtime main 8f68bc 1e2c1001 bd400120   # fmov s1, #0.5 -> ldr s0, [x9] (global ratio)
site realtime main 8f68cc 5e21d800 d503201f   # scvtf s0, s0 -> nop
site realtime main 8f68d0 1e210800 d503201f   # fmul s0, s0, s1 -> nop

# Slow-motion check (a trampoline): compares time object +0x28 * +0x38 (30 / time scale) with 100, the same value at 30 Hz
# and independent of the frame rate.
site bounce main ce3fec bd402900 bd402900 bd403901 1e210800   # ldr s0, [x8, #0x28]; ldr s1, [x8, #0x38]; fmul s0, s0, s1

# Hair and cloth: the actor animation update builds a second clock {seconds = factor / 30, ratio = factor} every rendered
# frame, so above 30 fps hair and cloth swing too fast (2x at 60). Three places have the update inlined and one calls the
# shared update (1.0.0 main+0x82c224); each inline place gets two replacements (at the fmul: seconds = measured seconds *
# factor; at the store of the factor into the clock's ratio slot: ratio = measured 30 Hz ratio * factor) and the call
# gets s2 / s3 scaled by the measured ratio. The sites are MacNativeGames' for 1.0.0 (game_patches.c, cloth_inline for the
# builds before 1.1.2), the words follow its veneers and totk-pc-recomp's patches-v121.txt (both MIT); the two places that
# store the pair with stp [x29, #-0x40] differ from the 1.2.1 table. Needs realtime (it reads the same page values).
site cloth main 7b5988 1e200900 f81f0fe9 @page:x9 bd400520 1e200900 f84107e9   # seconds: s0 = measured seconds * s8
site cloth main 7b598c 2d3823a0 d10083ff f90003e9 3d8007ff @page:x9 bd40013f 1e3f091f 2d387fa0 3dc007ff f94003e9 910083ff   # ratio: measured 30 Hz ratio * s8 into the slot
site cloth main 7b6b28 1e200900 f81f0fe9 @page:x9 bd400520 1e200900 f84107e9   # seconds: s0 = measured seconds * s8
site cloth main 7b6b2c 2d3823a0 d10083ff f90003e9 3d8007ff @page:x9 bd40013f 1e3f091f 2d387fa0 3dc007ff f94003e9 910083ff   # ratio: measured 30 Hz ratio * s8 into the slot
site cloth main 7f5ec4 1e210801 f81f0fe9 @page:x9 bd400521 1e210801 f84107e9   # seconds: s1 = measured seconds * s0
site cloth main 7f5ed0 bd0117e0 d10083ff f90003e9 3d8007ff @page:x9 bd40013f 1e3f081f bd0137ff 3dc007ff f94003e9 910083ff   # ratio: measured 30 Hz ratio * s0 into the slot
site cloth main 7b6358 9401d7b3 @page:x8 bd400104 1e240842 1e240863 9401d7b3   # the call: s2 = factor / 30, s3 = factor, times the measured ratio
